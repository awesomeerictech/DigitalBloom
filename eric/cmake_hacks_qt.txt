Final CMakeLists.txt (recommended)
cmake_minimum_required(VERSION 3.21)
project(qateriadev LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick Qml)

# main executable
qt_add_executable(qateriadev
    main.cpp
)

# --- Main module: embedded directly into exe ---
qt_add_qml_module(qateriadev
    URI qateriadev
    VERSION 1.0
    QML_FILES
        qml/Main.qml
    RESOURCE_PREFIX "/qateriadev"
    NO_PLUGIN
    NO_RESOURCE_TARGET_PATH   # valid because first arg is an existing executable target
)

# --- Components module: static module (embedded) ---
qt_add_qml_module(qateriadev_qml_components
    STATIC                    # <<-- important: produce a static module target
    URI qateriadev.Qml
    VERSION 1.0
    QML_FILES
        ${MY_QML_FILES}       # list all component QML/JS files
    RESOURCE_PREFIX "/qateriadev/Qml"
    NO_PLUGIN                 # prevent separate plugin target generation
)

# Link the static module into the executable so its resources & metadata are available
target_link_libraries(qateriadev
    PRIVATE
        Qt6::Quick
        Qt6::Qml
        qateriadev_qml_components
)

main.cpp (robust loader)
#include <QGuiApplication>
#include <QQmlApplicationEngine>

int main(int argc, char *argv[])
{
    QGuiApplication app(argc, argv);
    QQmlApplicationEngine engine;

    // Make sure the engine can resolve qrc:/ style imports for embedded modules:
    engine.addImportPath("qrc:/");

    // Preferred: load by module/entry (Qt 6.5+)
    engine.loadFromModule(QStringLiteral("qateriadev"), QStringLiteral("Main"));
    // Alternatively: engine.load(QUrl("qrc:/qateriadev/Main.qml"));

    if (engine.rootObjects().isEmpty())
        return -1;
    return app.exec();
}

Notes & checklist (to avoid the usual pitfalls)

STATIC is required for the components module so it becomes a static library target you can link to the exe.

NO_PLUGIN prevents CMake from generating a plugin target for the module — we don’t want separate DLLs.

NO_RESOURCE_TARGET_PATH is only used on the exe-target call (the first qt_add_qml_module where the first argument is the executable target name). Do not put it on the static module call.

Clean your build directory after changing these options (delete the build dir or git clean -fdx) — stale generated files commonly confuse CMake/Qt.

Verify resources with rcc --list path\to\qateriadev.exe (on Windows use findstr to filter). You should see entries like :/qateriadev/Main.qml and :/qateriadev/Qml/ButtonWireframe.qml and /qml/assets/....

Imports in QML use the URIs you specified:

import qateriadev 1.0
import qateriadev.Qml 1.0


If you ever need plugin initialization code (C++ registration inside a plugin), you would not use NO_PLUGIN and would then have to link the plugin target or use Q_IMPORT_QML_PLUGIN for static linking. But for pure QML/JS + resources, the above NO_PLUGIN + STATIC approach is simplest.
